> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A public hub site for Project IUoC — searchable, series-organized catalog of the channel's 659 YouTube videos, plus about/mission, socials, and donation links.

**Architecture:** Next.js App Router frontend reads from Convex via server components (`fetchQuery` from `convex/nextjs`) so content is in the initial HTML for SEO; interactive catalog browsing (search/filter/load-more) is a client island. **Tech Stack:** Next.js (App Router, from `nextjs-shadcn` Convex template), Convex (DB + cron + full-text search), Tailwind CSS v4 + shadcn/ui, next-themes, bun, vitest.

```ts
// convex/lib/youtube.ts
export type MergeableVideo = {
  youtubeId: string;
  slug: string;
  title: string;
  description: string;
  publishedAt: number;
  thumbnail: string;
  durationSeconds: number;
  playlistIds: string[];
  tags: string[];
  featured: boolean;
  searchText: string;
};

export function parseIso8601Duration(iso: string): number {
  const match = /^PT(?:(\d+)H)?(?:(\d+)M)?(?:(\d+)S)?$/.exec(iso);
  if (!match) return 0;
  const [, h, m, s] = match;
  return (Number(h ?? 0) * 3600) + (Number(m ?? 0) * 60) + Number(s ?? 0);
}

export function makeSlug(title: string, youtubeId: string): string {
  const base = title
    .toLowerCase()
    .normalize("NFKD")
    .replace(/[\u0300-\u036f]/g, "")
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "")
    .slice(0, 80)
    .replace(/-+$/g, "");
  const suffix = youtubeId.toLowerCase().replace(/[^a-z0-9]/g, "").slice(-4);
  return `${base.length > 0 ? base : "video"}-${suffix}`;
}

export function buildSearchText(title: string, description: string): string {
  return `${title}\n${description}`;
}

export function mergeVideo(
  existing: MergeableVideo,
  incoming: MergeableVideo,
): MergeableVideo {
  return {
    ...incoming,
    slug: existing.slug,
    tags: existing.tags,
    featured: existing.featured,
  };
}

export function chunk<T>(arr: T[], size: number): T[][] {
  const out: T[][] = [];
  for (let i = 0; i < arr.length; i += size) out.push(arr.slice(i, i + size));
  return out;
}
```

```ts
// convex/ingest.ts
import { v } from "convex/values";
import { internal } from "./_generated/api";
import { internalAction, internalMutation } from "./_generated/server";
import {
  buildSearchText,
  chunk,
  makeSlug,
  mergeVideo,
  parseIso8601Duration,
} from "./lib/youtube";

const API = "https://www.googleapis.com/youtube/v3";

const videoValidator = v.object({
  youtubeId: v.string(),
  slug: v.string(),
  title: v.string(),
  description: v.string(),
  publishedAt: v.number(),
  thumbnail: v.string(),
  durationSeconds: v.number(),
  playlistIds: v.array(v.string()),
  tags: v.array(v.string()),
  featured: v.boolean(),
  searchText: v.string(),
});

const playlistValidator = v.object({
  youtubeId: v.string(),
  title: v.string(),
  description: v.string(),
  thumbnail: v.string(),
  itemCount: v.number(),
});

async function ytFetch(
  path: string,
  params: Record<string, string>,
): Promise<any> {
  const key = process.env.YOUTUBE_API_KEY;
  if (!key) throw new Error("YOUTUBE_API_KEY is not set");
  const qs = new URLSearchParams({ ...params, key });
  const res = await fetch(`${API}/${path}?${qs}`);
  if (!res.ok) {
    throw new Error(`YouTube API ${path} failed: ${res.status} ${await res.text()}`);
  }
  return await res.json();
}

async function listAll(
  path: string,
  params: Record<string, string>,
): Promise<any[]> {
  const items: any[] = [];
  let pageToken: string | undefined;
  do {
    const page = await ytFetch(path, {
      ...params,
      maxResults: "50",
      ...(pageToken ? { pageToken } : {}),
    });
    items.push(...(page.items ?? []));
    pageToken = page.nextPageToken;
  } while (pageToken);
  return items;
}

function bestThumbnail(thumbnails: any): string {
  return (
    thumbnails?.maxres?.url ??
    thumbnails?.standard?.url ??
    thumbnails?.high?.url ??
    thumbnails?.medium?.url ??
    thumbnails?.default?.url ??
    ""
  );
}

export const run = internalAction({
  args: {},
  handler: async (ctx) => {
    const channelId = process.env.YOUTUBE_CHANNEL_ID!;
    const uploadsId = process.env.YOUTUBE_UPLOADS_PLAYLIST_ID!;

    // 1. All uploaded video ids
    const uploadItems = await listAll("playlistItems", {
      part: "contentDetails",
      playlistId: uploadsId,
    });
    const videoIds = uploadItems
      .map((i) => i.contentDetails?.videoId)
      .filter((id): id is string => typeof id === "string");

    // 2. Canonical video metadata in batches of 50
    const rawVideos: any[] = [];
    for (const ids of chunk(videoIds, 50)) {
      const page = await ytFetch("videos", {
        part: "snippet,contentDetails",
        id: ids.join(","),
      });
      rawVideos.push(...(page.items ?? []));
    }

    // 3. Playlists and their membership
    const rawPlaylists = await listAll("playlists", {
      part: "snippet,contentDetails",
      channelId,
    });
    const membership = new Map<string, string[]>();
    for (const p of rawPlaylists) {
      const items = await listAll("playlistItems", {
        part: "contentDetails",
        playlistId: p.id,
      });
      for (const item of items) {
        const vid = item.contentDetails?.videoId;
        if (!vid) continue;
        membership.set(vid, [...(membership.get(vid) ?? []), p.id]);
      }
    }

    const videos = rawVideos.map((raw) => ({
      youtubeId: raw.id,
      slug: makeSlug(raw.snippet.title, raw.id),
      title: raw.snippet.title as string,
      description: (raw.snippet.description ?? "") as string,
      publishedAt: Date.parse(raw.snippet.publishedAt),
      thumbnail: bestThumbnail(raw.snippet.thumbnails),
      durationSeconds: parseIso8601Duration(raw.contentDetails?.duration ?? ""),
      playlistIds: membership.get(raw.id) ?? [],
      tags: [],
      featured: false,
      searchText: buildSearchText(
        raw.snippet.title,
        raw.snippet.description ?? "",
      ),
    }));

    const playlists = rawPlaylists.map((p) => ({
      youtubeId: p.id as string,
      title: p.snippet.title as string,
      description: (p.snippet.description ?? "") as string,
      thumbnail: bestThumbnail(p.snippet.thumbnails),
      itemCount: (p.contentDetails?.itemCount ?? 0) as number,
    }));

    for (const batch of chunk(videos, 100)) {
      await ctx.runMutation(internal.ingest.upsertVideos, { videos: batch });
    }
    await ctx.runMutation(internal.ingest.upsertPlaylists, { playlists });

    console.log(`Ingested ${videos.length} videos, ${playlists.length} playlists`);
  },
});

export const upsertVideos = internalMutation({
  args: { videos: v.array(videoValidator) },
  handler: async (ctx, args) => {
    for (const video of args.videos) {
      const existing = await ctx.db
        .query("videos")
        .withIndex("by_youtubeId", (q) => q.eq("youtubeId", video.youtubeId))
        .unique();
      if (existing === null) {
        await ctx.db.insert("videos", video);
      } else {
        await ctx.db.patch(existing._id, mergeVideo(existing, video));
      }
    }
  },
});

export const upsertPlaylists = internalMutation({
  args: { playlists: v.array(playlistValidator) },
  handler: async (ctx, args) => {
    for (const playlist of args.playlists) {
      const existing = await ctx.db
        .query("playlists")
        .withIndex("by_youtubeId", (q) => q.eq("youtubeId", playlist.youtubeId))
        .unique();
      if (existing === null) {
        await ctx.db.insert("playlists", { ...playlist, visible: true });
      } else {
        await ctx.db.patch(existing._id, playlist);
      }
    }
  },
});
```

Failure semantics (per spec): any YouTube API error throws before the first mutation runs, so existing data is never partially clobbered; the error lands in Convex logs and the next cron run retries.

```ts
// convex/videos.ts
import { paginationOptsValidator } from "convex/server";
import { v } from "convex/values";
import { query } from "./_generated/server";

export const latest = query({
  args: { limit: v.number() },
  handler: async (ctx, args) => {
    return await ctx.db
      .query("videos")
      .withIndex("by_publishedAt")
      .order("desc")
      .take(Math.min(args.limit, 50));
  },
});

export const featured = query({
  args: {},
  handler: async (ctx) => {
    const all = await ctx.db
      .query("videos")
      .withIndex("by_publishedAt")
      .order("desc")
      .collect();
    return all.filter((video) => video.featured).slice(0, 6);
  },
});

export const bySlug = query({
  args: { slug: v.string() },
  handler: async (ctx, args) => {
    return await ctx.db
      .query("videos")
      .withIndex("by_slug", (q) => q.eq("slug", args.slug))
      .unique();
  },
});

export const byPlaylist = query({
  args: { playlistYoutubeId: v.string(), limit: v.optional(v.number()) },
  handler: async (ctx, args) => {
    const all = await ctx.db
      .query("videos")
      .withIndex("by_publishedAt")
      .order("desc")
      .collect();
    const matching = all.filter((video) =>
      video.playlistIds.includes(args.playlistYoutubeId),
    );
    return matching.slice(0, args.limit ?? 100);
  },
});

export const search = query({
  args: { query: v.string(), tag: v.optional(v.string()) },
  handler: async (ctx, args) => {
    const results = await ctx.db
      .query("videos")
      .withSearchIndex("by_searchText", (q) =>
        q.search("searchText", args.query),
      )
      .take(100);
    const filtered = args.tag
      ? results.filter((video) => video.tags.includes(args.tag!))
      : results;
    return filtered.slice(0, 30);
  },
});

export const paginate = query({
  args: { paginationOpts: paginationOptsValidator, tag: v.optional(v.string()) },
  handler: async (ctx, args) => {
    const page = await ctx.db
      .query("videos")
      .withIndex("by_publishedAt")
      .order("desc")
      .paginate(args.paginationOpts);
    if (!args.tag) return page;
    return {
      ...page,
      page: page.page.filter((video) => video.tags.includes(args.tag!)),
    };
  },
});

export const allTags = query({
  args: {},
  handler: async (ctx) => {
    const all = await ctx.db.query("videos").collect();
    const tags = new Set<string>();
    for (const video of all) for (const tag of video.tags) tags.add(tag);
    return [...tags].sort();
  },
});

export const allSlugs = query({
  args: {},
  handler: async (ctx) => {
    const all = await ctx.db.query("videos").collect();
    return all.map((video) => ({
      slug: video.slug,
      publishedAt: video.publishedAt,
    }));
  },
});
```

```tsx
export function formatDuration(totalSeconds: number): string {
  const h = Math.floor(totalSeconds / 3600);
  const m = Math.floor((totalSeconds % 3600) / 60);
  const s = totalSeconds % 60;
  const mm = h > 0 ? String(m).padStart(2, "0") : String(m);

}
```
